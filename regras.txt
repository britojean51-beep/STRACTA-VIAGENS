rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // ---- quem é quem ----
    function meuEmail() {
      return request.auth != null ? request.auth.token.email.lower() : '';
    }
    function meuDoc() {
      return get(/databases/$(database)/documents/usuarios/$(meuEmail())).data;
    }
    // Está logado, tem liberação e não foi desativado.
    // Usa .get(campo, padrão): ler um campo que não existe dá ERRO nas regras,
    // e erro nega tudo — então nunca acessamos o campo direto.
    function liberado() {
      return request.auth != null
        && exists(/databases/$(database)/documents/usuarios/$(meuEmail()))
        && meuDoc().get('ativo', true) != false;
    }
    function ehGestor() {
      return liberado() && meuDoc().get('perfil', '') == 'gestor';
    }
    // O desenvolvedor: o acesso dele não se mexe, nem por outro gestor.
    function contaDoDono(email) { return email == 'jean@gp2t.local'; }
    function ehDono() { return meuEmail() == 'jean@gp2t.local'; }

    // ---- quem pode entrar no app ----
    // Cada um lê a própria liberação. Só o gestor lê a lista e cria/remove.
    // O acesso do desenvolvedor só ele altera ou exclui.
    // No update, qualquer um arruma o PRÓPRIO nome — sem poder se promover a
    // gestor nem se desbloquear, porque perfil e ativo têm que continuar iguais.
    match /usuarios/{email} {
      allow get: if request.auth != null && (meuEmail() == email || ehGestor());
      allow list: if ehGestor();
      allow create, delete: if ehGestor() && (!contaDoDono(email) || ehDono());
      allow update: if (ehGestor() && (!contaDoDono(email) || ehDono()))
        || (meuEmail() == email
            && request.resource.data.get('perfil', '') == resource.data.get('perfil', '')
            && request.resource.data.get('ativo', true) == resource.data.get('ativo', true));
    }

    // ---- os dados da frota ----
    match /frota/{frota} {
      allow read: if liberado();
      allow write: if ehGestor();

      // Lançamentos: todo mundo liberado lê e lança.
      // Corrigir ou apagar: só o gestor ou quem lançou.
      // (a trava de "só o dia de hoje" fica no app; aqui o que protege é o dono)
      match /{tipo}/{id} {
        allow read: if liberado()
          && (tipo == 'abastecimentos' || tipo == 'viagens' || tipo == 'manutencoes');
        allow create: if liberado()
          && (tipo == 'abastecimentos' || tipo == 'viagens' || tipo == 'manutencoes')
          && request.resource.data.criadoPor == meuEmail();
        allow update, delete: if liberado()
          && (tipo == 'abastecimentos' || tipo == 'viagens' || tipo == 'manutencoes')
          && (ehGestor() || resource.data.criadoPor == meuEmail());
      }

      // Cadastro da frota (equipamentos, operadores, metas): só o gestor muda.
      match /cadastro/frota {
        allow read: if liberado();
        allow write: if ehGestor();
      }
      // Operação (último KM/horímetro e status) e tanques: o operador atualiza ao lançar.
      match /cadastro/operacao {
        allow read, write: if liberado();
      }
      match /cadastro/estoque {
        allow read, write: if liberado();
      }
    }

    // ---- Checklist de máquinas e caminhões (gp2t-checklist.web.app) ----
    // Quem entra: documento ativo em <base>usuarios. Funções:
    //   motorista, operador -> fazem checklist e veem só os seus
    //   tecnico (segurança) -> + veem todos e marcam "visualizado"; não resolvem
    //   gestor              -> + veem todos, resolvem (tratativa) e mexem em usuários,
    //                          equipamentos e modelos (não nomeiam administrador)
    //   admin               -> + apagam checklists e nomeiam administradores
    //
    // VÁRIAS EMPRESAS: as mesmas regras valem para cada empresa, com os dados dela.
    //   e == 'stracta'  -> coleções checklist_* (a primeira empresa, onde sempre esteve)
    //   outra empresa e -> empresas/{e}/<coleção>
    // Cada empresa só enxerga a própria: o cadastro da pessoa é conferido DENTRO da
    // empresa (empresas/{e}/usuarios). Só o dono cria e suspende empresas.
    // O gestor do app de frota entra só na STRACTA, como admin — é quem libera os primeiros.
    function ckRef(e, col, id) {
      return path('/databases/' + database + '/documents/'
        + (e == 'stracta' ? 'checklist_' + col : 'empresas/' + e + '/' + col) + '/' + id);
    }
    // empresa suspensa pelo dono: ninguém dela entra (os dados ficam guardados)
    function ckEmpresaAtiva(e) {
      return e == 'stracta' || get(/databases/$(database)/documents/empresas/$(e)).data.get('ativo', true) != false;
    }
    // A função da pessoa na empresa e ('' = não entra). Lida uma vez só por regra:
    // ler o cadastro em cada teste estoura o limite de expressões do Firestore.
    function ckPerfil(e) {
      let ref = ckRef(e, 'usuarios', meuEmail());
      let cad = request.auth != null && exists(ref)
        && get(ref).data.get('ativo', true) != false && ckEmpresaAtiva(e);
      let p = cad ? get(ref).data.get('perfil', 'motorista') : '';
      return ckFrota(e) ? 'admin' : (cad && p == '' ? 'motorista' : p);
    }
    function ckFrota(e) { return e == 'stracta' && ehGestor(); }
    function ckCadastrado(e) { return ckPerfil(e) != ''; }
    function ckLiberado(e) { return ckPerfil(e) != ''; }
    function ckAdmin(e) { return ckPerfil(e) == 'admin'; }
    function ckGestao(e) { return ckPerfil(e) in ['admin', 'gestor']; }
    function ckVeTodos(e) { return ckPerfil(e) in ['admin', 'gestor', 'tecnico']; }

    // Usuários: admin e gestor cadastram, mudam função e bloqueiam; administrador
    // só o administrador nomeia, muda ou bloqueia. Em outra empresa, o login tem que
    // ser dela (usuario@<e>.gp2t.local) e o dono cria o primeiro administrador.
    function ckUsGet(e, email) {
      return request.auth != null && ((meuEmail() == email && ckEmpresaAtiva(e)) || ckGestao(e));
    }
    function ckUsDaEmpresa(e, email) { return e == 'stracta' || email.matches('[^@]+@' + e + '[.]gp2t[.]local'); }
    function ckUsCreate(e, email) {
      let p = ckPerfil(e);
      return ckUsDaEmpresa(e, email) && (p == 'admin' || (p == 'gestor' && request.resource.data.get('perfil', 'motorista') != 'admin'));
    }
    function ckUsUpdate(e) {
      let p = ckPerfil(e);
      return p == 'admin' || (p == 'gestor' && resource.data.get('perfil', '') != 'admin'
        && request.resource.data.get('perfil', 'motorista') != 'admin');
    }
    // Equipamento: o cadastro é do gestor e do admin. O campo "pendencias" (defeitos em
    // aberto, de onde sai a situação Operacional/Restrição/Manutenção):
    // quem faz checklist só ACRESCENTA; tirar (resolvido -> volta a operar)
    // é do gestor e do admin.
    function ckEqUpdate(e) {
      let p = ckPerfil(e);
      return p in ['admin', 'gestor']
        || (p != ''
            && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['pendencias'])
            && pendenciaSoSobe(resource.data.get('pendencias', {}), request.resource.data.get('pendencias', {})));
    }
    // Quem faz checklist só pode PIORAR a situação: acrescentar pendência nova ou
    // subir uma de "restricao" para "bloqueado" (problema apontado depois que obriga
    // parada). Tirar ou rebaixar (bloqueado -> restricao) é só da gestão.
    function pendenciaSoSobe(antes, depois) {
      return depois.keys().hasAll(antes.keys())
        && depois.values().hasOnly(['bloqueado', 'restricao'])
        && depois.diff(antes).changedKeys().size() <= 1
        && depois.values().removeAll(['restricao']).size() >= antes.values().removeAll(['restricao']).size();
    }
    // O checklist feito: quem fez lê o seu; gestor e admin leem todos.
    // Depois de enviado, quem fez só mexe em "midias" e "pendentes" (marcar a
    // foto como enviada) e acrescenta "apontamentos" (problema que apareceu
    // durante o serviço) — as respostas originais não mudam mais. O gestor só mexe na
    // resolução (tratativa, resolucoes e solicitacoes por item e as fotos). Apagar: só o admin.
    // resource == null: checklist que não existe (foi apagado) responde "não
    // existe" em vez de "sem permissão" — é assim que o celular de quem fez
    // tira da lista "Meus últimos checklists". Não revela nada.
    function ckInspRead(e) {
      let p = ckPerfil(e);
      return p in ['admin', 'gestor', 'tecnico'] || (p != '' && (resource == null || resource.data.criadoPor == meuEmail()));
    }
    // criar: em nome próprio e sem já vir "resolvido", "visto" ou com apontamento
    function ckInspCreate(e) {
      return ckLiberado(e) && request.resource.data.criadoPor == meuEmail()
        && !request.resource.data.keys().hasAny(['resolucoes', 'solicitacoes', 'vistos', 'apontamentos']);
    }
    function ckInspUpdate(e) {
      let p = ckPerfil(e);
      return p == 'admin'
        || (p == 'gestor' && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['tratativa', 'resolucoes', 'solicitacoes', 'midias', 'pendentes', 'vistos']))
        || (p != '' && resource.data.criadoPor == meuEmail()
            && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['midias', 'pendentes', 'apontamentos'])
            // problema apontado durante o serviço: só acrescenta, nunca tira um apontamento
            && request.resource.data.get('apontamentos', {}).keys().hasAll(resource.data.get('apontamentos', {}).keys()))
        || (p == 'tecnico' && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['vistos']));
    }
    // Fotos do checklist (enquanto o Storage não estiver ligado): um documento
    // por foto. Mesma regra de quem vê o checklist; só imagem JPEG, até ~950 KB.
    // Quem fez o checklist também vê as fotos da correção (tiradas pelo gestor):
    // é como o motorista acompanha a manutenção do caminhão dele.
    function fotoValida() {
      return request.resource.data.criadoPor == meuEmail()
        && request.resource.data.dados is string
        && request.resource.data.dados.size() < 950000
        && request.resource.data.dados.matches('data:image/jpeg;base64,.*');
    }
    function ckFotoRead(e) {
      let p = ckPerfil(e);
      return p in ['admin', 'gestor', 'tecnico'] || (p != '' && (resource.data.criadoPor == meuEmail()
        || get(ckRef(e, 'inspecoes', resource.data.inspecao)).data.get('criadoPor', '') == meuEmail()));
    }
    // a foto é a prova: ninguém troca a foto de outra pessoa (update só de quem enviou)
    function ckFotoUpdate(e) { return ckLiberado(e) && resource.data.criadoPor == meuEmail() && fotoValida(); }

    // -- STRACTA: coleções checklist_* --
    match /checklist_usuarios/{email} {
      allow get: if ckUsGet('stracta', email);
      allow list: if ckGestao('stracta');
      allow create: if ckUsCreate('stracta', email);
      allow update: if ckUsUpdate('stracta');
      allow delete: if ckAdmin('stracta');
    }
    match /checklist_equipamentos/{id} {
      allow read: if ckLiberado('stracta');
      allow create, delete: if ckGestao('stracta');
      allow update: if ckEqUpdate('stracta');
    }
    // Lubrificação e inspeção: a gestão lança o que foi feito em cada equipamento.
    // O resumo (última lubrificação/inspeção e a próxima marcada) fica no equipamento,
    // que todos leem; o histórico, só quem vê todos os checklists.
    match /checklist_servicos/{id} {
      allow read: if ckVeTodos('stracta');
      allow create: if ckGestao('stracta') && request.resource.data.porEmail == meuEmail();
      allow update, delete: if ckGestao('stracta');
    }
    match /checklist_modelos/{id} {
      allow read: if ckLiberado('stracta');
      allow write: if ckGestao('stracta');
    }
    match /checklist_inspecoes/{id} {
      allow read: if ckInspRead('stracta');
      allow create: if ckInspCreate('stracta');
      allow update: if ckInspUpdate('stracta');
      allow delete: if ckAdmin('stracta');
    }
    match /checklist_fotos/{id} {
      allow read: if ckFotoRead('stracta');
      allow create: if ckLiberado('stracta') && fotoValida();
      allow update: if ckFotoUpdate('stracta');
      allow delete: if ckAdmin('stracta');
    }

    // -- outras empresas: empresas/{e}/... (mesmas regras, cada uma na sua) --
    // A ficha da empresa: quem é dela lê; só o dono cria, muda ou suspende.
    // 'stracta' é reservado (é a primeira empresa, nas coleções checklist_*).
    match /empresas/{e} {
      allow read: if ehDono() || ckCadastrado(e);
      allow create: if ehDono() && e != 'stracta' && e.matches('[a-z0-9][a-z0-9-]{1,19}');
      allow update: if ehDono();
      allow delete: if false;

      match /usuarios/{email} {
        allow get: if ehDono() || ckUsGet(e, email);
        allow list: if ehDono() || ckGestao(e);
        allow create: if (ehDono() && ckUsDaEmpresa(e, email)) || ckUsCreate(e, email);
        allow update: if ehDono() || ckUsUpdate(e);
        allow delete: if ckAdmin(e);
      }
      match /equipamentos/{id} {
        allow read: if ckLiberado(e);
        allow create, delete: if ckGestao(e);
        allow update: if ckEqUpdate(e);
      }
      match /servicos/{id} {
        allow read: if ckVeTodos(e);
        allow create: if ckGestao(e) && request.resource.data.porEmail == meuEmail();
        allow update, delete: if ckGestao(e);
      }
      match /modelos/{id} {
        allow read: if ckLiberado(e);
        allow write: if ckGestao(e);
      }
      match /inspecoes/{id} {
        allow read: if ckInspRead(e);
        allow create: if ckInspCreate(e);
        allow update: if ckInspUpdate(e);
        allow delete: if ckAdmin(e);
      }
      match /fotos/{id} {
        allow read: if ckFotoRead(e);
        allow create: if ckLiberado(e) && fotoValida();
        allow update: if ckFotoUpdate(e);
        allow delete: if ckAdmin(e);
      }
    }

    // Nada além disso.
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
