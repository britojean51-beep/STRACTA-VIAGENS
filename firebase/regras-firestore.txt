rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // Quem está logado lê a própria liberação (documento com o e-mail dele).
    // Só gestor lê a lista toda e cria/edita/remove liberações.
    match /usuarios/{email} {
      function ehGestor() {
        return request.auth != null
          && exists(/databases/$(database)/documents/usuarios/$(request.auth.token.email.lower()))
          && get(/databases/$(database)/documents/usuarios/$(request.auth.token.email.lower())).data.perfil == 'gestor';
      }
      allow get: if request.auth != null
                 && (request.auth.token.email.lower() == email || ehGestor());
      allow list, create, update, delete: if ehGestor();
    }

    // Nada mais é liberado por enquanto (os lançamentos ficam no aparelho
    // e na planilha do Google Sheets). Na Etapa 2, os dados entram aqui.
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
